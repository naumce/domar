# Domar Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A running multi-tenant monorepo where a manager registers a company, logs in, and manages tax rates, contacts, buildings and units through a typed API and a Vue 3 web app, with the Macedonian country pack and the money primitives every later plan depends on.

**Architecture:** pnpm workspace with a pure `packages/domain` (money, country packs), `packages/db` (Drizzle schema + migrations on Postgres 16), `apps/api` (Fastify 5, zod-validated routes, cookie sessions, tenant context on every query) and `apps/web` (Vue 3 + Vite + Pinia + vue-i18n + Tailwind). Every tenant-scoped table has `tenant_id`; repositories take a `TenantContext` first.

**Tech Stack:** Node 24, pnpm 10, TypeScript 5 strict ESM, Vitest, Drizzle ORM + drizzle-kit, postgres.js, Fastify 5, fastify-type-provider-zod, zod, @fastify/cookie, @fastify/rate-limit, Vue 3, Vite 6, vue-router 4, Pinia, vue-i18n 10, Tailwind 4, Playwright.

**Spec:** `docs/superpowers/specs/2026-09-21-domar-phase1-design.md`

## Global Constraints

- Node `>=24`, pnpm `>=10`, TypeScript `strict: true`, ESM only (`"type": "module"` in every package).
- Postgres 16; schema must run unchanged on Supabase Postgres.
- Every tenant-scoped table has `tenant_id uuid not null` and an index whose first column is `tenant_id`.
- Money columns are `bigint` with suffix `_minor`; never float. In TypeScript money is `number` (minor units).
- All web strings go through vue-i18n; Macedonian (`mk`) first, English (`en`) second.
- No secrets in the repo; `.env.example` documents every variable.
- Commit after every green test cycle with conventional commit messages (`feat:`, `test:`, `chore:`).
- Run commands from the repo root `/Users/naum/Developer/personal/domar` unless a step says otherwise.

```ts
/** Round half away from zero to an integer. */
export function roundHalfUp(value: number): number {
  const sign = value < 0 ? -1 : 1;
  return sign * Math.floor(Math.abs(value) + 0.5 + Number.EPSILON);
}

/** price (minor units) × decimal quantity, rounded half up to minor units. */
export function mulMinor(priceMinor: number, quantity: number): number {
  return roundHalfUp(priceMinor * quantity);
}

/** Apply a tax rate in basis points (1800 = 18%). */
export function applyTaxBp(netMinor: number, rateBp: number): { taxMinor: number; grossMinor: number } {
  const taxMinor = roundHalfUp((netMinor * rateBp) / 10_000);
  return { taxMinor, grossMinor: netMinor + taxMinor };
}

/**
 * Split totalMinor across weights so that parts sum exactly to totalMinor.
 * Largest-remainder method: floor each share, then hand out the leftover units
 * to the shares with the largest fractional parts (ties broken by index).
 */
export function allocateLargestRemainder(totalMinor: number, weights: number[]): number[] {
  if (!Number.isInteger(totalMinor)) throw new RangeError('totalMinor must be an integer');
  const sumW = weights.reduce((a, b) => a + b, 0);
  if (sumW === 0) {
    if (totalMinor === 0) return weights.map(() => 0);
    throw new RangeError('cannot allocate a non-zero total over zero weights');
  }
  const sign = totalMinor < 0 ? -1 : 1;
  const abs = Math.abs(totalMinor);
  const raw = weights.map((w) => (abs * w) / sumW);
  const floors = raw.map((r) => Math.floor(r));
  let remainder = abs - floors.reduce((a, b) => a + b, 0);
  const order = raw
    .map((r, i) => ({ i, frac: r - Math.floor(r) }))
    .sort((a, b) => b.frac - a.frac || a.i - b.i);
  for (const { i } of order) {
    if (remainder <= 0) break;
    if (weights[i] === 0) continue;
    floors[i] = (floors[i] ?? 0) + 1;
    remainder -= 1;
  }
  return floors.map((f) => sign * f);
}
```

```ts
import type { CountryPack } from './types.js';

function groupThousands(intPart: string, sep: string): string {
  return intPart.replace(/\B(?=(\d{3})+(?!\d))/g, sep);
}

export const mkPack: CountryPack = {
  code: 'mk',
  currency: 'MKD',
  locale: 'mk-MK',
  defaultTaxRates: [
    { name: 'Без ДДВ', rateBp: 0, isDefault: false },
    { name: 'ДДВ 5%', rateBp: 500, isDefault: false },
    { name: 'ДДВ 10%', rateBp: 1000, isDefault: false },
    { name: 'ДДВ 18%', rateBp: 1800, isDefault: true },
  ],
  invoiceLayout: {
    showBuildingWideAmountPerLine: true,
    showManagementFeeLine: true,
    showReserveFundContribution: true,
    showReserveFundBuildingBalance: true,
    showPreviousBalance: true,
  },
  reminderRules: {
    consecutiveUnpaid: 2,
    totalUnpaid: 3,
    feeAllowed: false,
    channels: ['email', 'sms', 'letter'],
  },
  formatMoney(minor: number): string {
    const sign = minor < 0 ? '-' : '';
    const abs = Math.abs(minor);
    const intPart = Math.floor(abs / 100).toString();
    const frac = (abs % 100).toString().padStart(2, '0');
    return `${sign}${groupThousands(intPart, '.')},${frac} ден.`;
  },
  parseAmountToMinor(text: string): number {
    const cleaned = text.replace(/\s/g, '').replace(/ден\.?/i, '');
    if (!/^-?[\d.,]+$/.test(cleaned)) throw new Error(`not an amount: ${text}`);
    const lastComma = cleaned.lastIndexOf(',');
    const lastDot = cleaned.lastIndexOf('.');
    let normalized: string;
    if (lastComma > lastDot) {
      normalized = cleaned.replace(/\./g, '').replace(',', '.');
    } else if (lastDot > lastComma) {
      const decimals = cleaned.length - lastDot - 1;
      normalized = decimals === 3 && lastComma === -1 ? cleaned.replace(/\./g, '') : cleaned.replace(/,/g, '');
    } else {
      normalized = cleaned;
    }
    const value = Number(normalized);
    if (!Number.isFinite(value)) throw new Error(`not an amount: ${text}`);
    return Math.round(value * 100);
  },
};
```

```ts
export async function runMigrations(url: string): Promise<void> {
  const sql = postgres(url, { max: 1, onnotice: () => {} });
```

```ts
export async function createTestDatabase(): Promise<{ url: string; db: Db; sql: postgres.Sql; close(): Promise<void> }> {
  const name = `domar_test_${randomBytes(6).toString('hex')}`;

  };
```

```ts
import cookie from '@fastify/cookie';
import cors from '@fastify/cors';
import rateLimit from '@fastify/rate-limit';
import type { Db } from '@domar/db';
import Fastify, { type FastifyInstance } from 'fastify';
import { hasZodFastifySchemaValidationErrors, serializerCompiler, validatorCompiler, type ZodTypeProvider } from 'fastify-type-provider-zod';
import type postgres from 'postgres';
import type { Config } from './config.js';
import { AppError } from './errors.js';
import { dbPlugin } from './plugins/db.js';
import { healthRoutes } from './routes/health.js';

export async function buildApp(opts: { db: Db; sql: postgres.Sql; config: Config }): Promise<FastifyInstance> {
  const app = Fastify({ logger: opts.config.env !== 'test', trustProxy: true }).withTypeProvider<ZodTypeProvider>();
  app.setValidatorCompiler(validatorCompiler);
  app.setSerializerCompiler(serializerCompiler);

  await app.register(cors, { origin: opts.config.corsOrigins, credentials: true });
  await app.register(cookie);
  await app.register(rateLimit, { global: false });
  await app.register(dbPlugin, { db: opts.db, sql: opts.sql });
  app.decorate('config', opts.config);

  app.setNotFoundHandler((req, reply) => {
    reply.status(404).send({ error: { code: 'not_found', message: `Route ${req.method} ${req.url} not found` } });
  });

  app.setErrorHandler((err, _req, reply) => {
    if (err instanceof AppError) {
      return reply.status(err.statusCode).send({ error: { code: err.code, message: err.message, ...(err.details !== undefined ? { details: err.details } : {}) } });
    }
    if (hasZodFastifySchemaValidationErrors(err)) {
      return reply.status(400).send({ error: { code: 'validation', message: 'Invalid request', details: err.validation } });
    }
    if (typeof (err as { statusCode?: number }).statusCode === 'number' && (err as { statusCode: number }).statusCode < 500) {
      const e = err as { statusCode: number; code?: string; message: string };
      return reply.status(e.statusCode).send({ error: { code: e.code?.toLowerCase() ?? 'error', message: e.message } });
    }
    app.log.error(err);
    return reply.status(500).send({ error: { code: 'internal', message: 'Internal error' } });
  });

  await app.register(async (api) => {
    await api.register(healthRoutes);
  }, { prefix: '/api' });

  return app;
}

declare module 'fastify' {
  interface FastifyInstance { config: Config }
}
```

```ts
import { randomBytes, scrypt, timingSafeEqual } from 'node:crypto';
import { promisify } from 'node:util';

const scryptAsync = promisify(scrypt) as (pw: string, salt: Buffer, len: number, opts: { N: number; r: number; p: number; maxmem: number }) => Promise<Buffer>;
const N = 16384, R = 8, P = 1, LEN = 64;

export async function hashPassword(plain: string): Promise<string> {
  const salt = randomBytes(16);
  const hash = await scryptAsync(plain, salt, LEN, { N, r: R, p: P, maxmem: 64 * 1024 * 1024 });
  return `scrypt$${N}$${salt.toString('base64url')}$${hash.toString('base64url')}`;
}

export async function verifyPassword(plain: string, stored: string): Promise<boolean> {
  const [algo, nStr, saltB64, hashB64] = stored.split('$');
  if (algo !== 'scrypt' || !nStr || !saltB64 || !hashB64) return false;
  const expected = Buffer.from(hashB64, 'base64url');
  const actual = await scryptAsync(plain, Buffer.from(saltB64, 'base64url'), expected.length, { N: Number(nStr), r: R, p: P, maxmem: 64 * 1024 * 1024 });
  return actual.length === expected.length && timingSafeEqual(actual, expected);
}
```

```ts
import { createHash, randomBytes } from 'node:crypto';
import { type Db, schema } from '@domar/db';
import { and, eq, gt } from 'drizzle-orm';

export const SESSION_COOKIE = 'domar_session';
const SESSION_DAYS = 14;

function hashToken(token: string): string {
  return createHash('sha256').update(token).digest('hex');
}

export async function createSession(db: Db, userId: string, meta: { userAgent?: string | undefined; ip?: string | undefined }): Promise<{ token: string; expiresAt: Date }> {
  const token = randomBytes(32).toString('base64url');
  const expiresAt = new Date(Date.now() + SESSION_DAYS * 24 * 3600 * 1000);
  await db.insert(schema.sessions).values({ tokenHash: hashToken(token), userId, expiresAt, userAgent: meta.userAgent ?? null, ip: meta.ip ?? null });
  return { token, expiresAt };
}

export async function resolveSession(db: Db, token: string) {
  const rows = await db
    .select({ user: schema.users, tenant: schema.tenants })
    .from(schema.sessions)
    .innerJoin(schema.users, eq(schema.users.id, schema.sessions.userId))
    .innerJoin(schema.tenants, eq(schema.tenants.id, schema.users.tenantId))
    .where(and(eq(schema.sessions.tokenHash, hashToken(token)), gt(schema.sessions.expiresAt, new Date())))
    .limit(1);
  return rows[0] ?? null;
}

export async function deleteSession(db: Db, token: string): Promise<void> {
  await db.delete(schema.sessions).where(eq(schema.sessions.tokenHash, hashToken(token)));
}
```

```ts
  app.decorateRequest('session', null);
```

```ts
export function requireRole(...roles: UserRole[]) {

    if (!roles.includes(req.ctx.role)) throw forbidden();
```

```ts
async function assertContactInTenant(db: Db, ctx: TenantContext, contactId: string | null | undefined) {
  if (!contactId) return;
  const c = await db.query.contacts.findFirst({ where: and(eq(schema.contacts.tenantId, ctx.tenantId), eq(schema.contacts.id, contactId)), columns: { id: true } });
  if (!c) throw badRequest('invalid_reference', 'representativeContactId does not belong to this organization');

export async function listBuildings(db: Db, ctx: TenantContext) {
  const rows = await db
    .select({ building: B, unitCount: sql<number>`(select count(*) from ${schema.units} u where u.building_id = ${B.id})`.mapWith(Number) })
    .from(B).where(eq(B.tenantId, ctx.tenantId)).orderBy(asc(B.name));
  return { rows: rows.map((r) => ({ ...r.building, unitCount: r.unitCount })) };
}

  const row = await db.query.buildings.findFirst({ where: and(eq(B.tenantId, ctx.tenantId), eq(B.id, id)) });
```

```ts
import { type Db, schema } from '@domar/db';
import { and, asc, eq } from 'drizzle-orm';
import { writeAudit } from '../../audit.js';
import { badRequest, conflict, notFound } from '../../errors.js';
import type { TenantContext } from '../../tenant-context.js';
import { getBuilding } from '../buildings/repo.js';
import type { UnitCreate, UnitPatch } from './schema.js';

const U = schema.units;
type Row = typeof U.$inferSelect;

function toOut(row: Row, owner: { id: string; firstName: string; lastName: string } | null) {
  return { ...row, areaM2: row.areaM2 === null ? null : Number(row.areaM2), ownershipShare: row.ownershipShare === null ? null : Number(row.ownershipShare), owner };
}

function toRow(input: Partial<UnitCreate>) {
  const { areaM2, ownershipShare, ...rest } = input;
  return {
    ...rest,
    ...(areaM2 !== undefined ? { areaM2: areaM2 === null ? null : areaM2.toString() } : {}),
    ...(ownershipShare !== undefined ? { ownershipShare: ownershipShare === null ? null : ownershipShare.toString() } : {}),
  };
}

async function validateRefs(db: Db, ctx: TenantContext, buildingId: string, input: Partial<UnitCreate>, selfId?: string) {
  if (input.ownerContactId) {
    const c = await db.query.contacts.findFirst({ where: and(eq(schema.contacts.tenantId, ctx.tenantId), eq(schema.contacts.id, input.ownerContactId)), columns: { id: true } });
    if (!c) throw badRequest('invalid_reference', 'ownerContactId does not belong to this organization');
  }
  if (input.attachedToUnitId) {
    if (input.attachedToUnitId === selfId) throw badRequest('invalid_attachment', 'a unit cannot be attached to itself');
    const target = await db.query.units.findFirst({ where: and(eq(U.tenantId, ctx.tenantId), eq(U.buildingId, buildingId), eq(U.id, input.attachedToUnitId)) });
    if (!target || !['apartment', 'commercial'].includes(target.kind)) throw badRequest('invalid_attachment', 'attachedToUnitId must be an apartment or commercial unit in the same building');
  }
}

async function ownerOf(db: Db, ownerContactId: string | null) {
  if (!ownerContactId) return null;
  const c = await db.query.contacts.findFirst({ where: eq(schema.contacts.id, ownerContactId), columns: { id: true, firstName: true, lastName: true } });
  return c ?? null;
}

export async function listUnits(db: Db, ctx: TenantContext, buildingId: string) {
  await getBuilding(db, ctx, buildingId);
  const rows = await db
    .select({ unit: U, owner: { id: schema.contacts.id, firstName: schema.contacts.firstName, lastName: schema.contacts.lastName } })
    .from(U).leftJoin(schema.contacts, eq(schema.contacts.id, U.ownerContactId))
    .where(and(eq(U.tenantId, ctx.tenantId), eq(U.buildingId, buildingId))).orderBy(asc(U.floor), asc(U.label));
  return { rows: rows.map((r) => toOut(r.unit, r.owner?.id ? r.owner : null)) };
}

export async function getUnit(db: Db, ctx: TenantContext, buildingId: string, id: string) {
  const row = await db.query.units.findFirst({ where: and(eq(U.tenantId, ctx.tenantId), eq(U.buildingId, buildingId), eq(U.id, id)) });
  if (!row) throw notFound('Unit');
  return row;
}

export async function createUnit(db: Db, ctx: TenantContext, buildingId: string, input: UnitCreate) {
  return db.transaction(async (tx) => {
    await getBuilding(tx, ctx, buildingId);
    await validateRefs(tx, ctx, buildingId, input);
    const dup = await tx.query.units.findFirst({ where: and(eq(U.buildingId, buildingId), eq(U.label, input.label)), columns: { id: true } });
    if (dup) throw conflict('label_taken', `Unit ${input.label} already exists in this building`);
    const [row] = await tx.insert(U).values({ ...toRow(input), label: input.label, kind: input.kind, tenantId: ctx.tenantId, buildingId }).returning();
    await writeAudit(tx, ctx, { action: 'unit.create', entityType: 'unit', entityId: row!.id, after: row });
    return toOut(row!, await ownerOf(tx, row!.ownerContactId));
  });
}

export async function updateUnit(db: Db, ctx: TenantContext, buildingId: string, id: string, patch: UnitPatch) {
  return db.transaction(async (tx) => {
    const before = await getUnit(tx, ctx, buildingId, id);
    await validateRefs(tx, ctx, buildingId, patch, id);
    if (patch.label && patch.label !== before.label) {
      const dup = await tx.query.units.findFirst({ where: and(eq(U.buildingId, buildingId), eq(U.label, patch.label)), columns: { id: true } });
      if (dup) throw conflict('label_taken', `Unit ${patch.label} already exists in this building`);
    }
    const [row] = await tx.update(U).set({ ...toRow(patch), updatedAt: new Date() }).where(and(eq(U.tenantId, ctx.tenantId), eq(U.id, id))).returning();
    await writeAudit(tx, ctx, { action: 'unit.update', entityType: 'unit', entityId: id, before, after: row });
    return toOut(row!, await ownerOf(tx, row!.ownerContactId));
  });
}

export async function deleteUnit(db: Db, ctx: TenantContext, buildingId: string, id: string) {
  return db.transaction(async (tx) => {
    const before = await getUnit(tx, ctx, buildingId, id);
    await tx.delete(U).where(and(eq(U.tenantId, ctx.tenantId), eq(U.id, id)));
    await writeAudit(tx, ctx, { action: 'unit.delete', entityType: 'unit', entityId: id, before });
  });
}
```

```ts
export class ApiError extends Error {
  constructor(public readonly status: number, public readonly code: string, message: string, public readonly details?: unknown) {
    super(message);
    this.name = 'ApiError';
  }
}

const BASE = (import.meta.env.VITE_API_URL as string | undefined) ?? 'http://localhost:3001';
type Query = Record<string, string | number | boolean | undefined>;

function url(path: string, query?: Query): string {
  const u = new URL(path, BASE);
  if (query) for (const [k, v] of Object.entries(query)) if (v !== undefined) u.searchParams.set(k, String(v));
  return u.toString();
}

async function request<T>(method: string, path: string, body?: unknown, query?: Query): Promise<T> {
  const res = await fetch(url(path, query), {
    method,
    credentials: 'include',
    headers: body !== undefined ? { 'content-type': 'application/json' } : {},
    body: body !== undefined ? JSON.stringify(body) : undefined,
  });
  if (res.status === 204) return undefined as T;
  const isJson = res.headers.get('content-type')?.includes('application/json');
  const data = isJson ? await res.json() : undefined;
  if (!res.ok) {
    const e = (data as { error?: { code?: string; message?: string; details?: unknown } } | undefined)?.error;
    throw new ApiError(res.status, e?.code ?? 'error', e?.message ?? res.statusText, e?.details);
  }
  return data as T;
}

export const api = {
  get: <T>(path: string, query?: Query) => request<T>('GET', path, undefined, query),
  post: <T>(path: string, body?: unknown) => request<T>('POST', path, body),
  patch: <T>(path: string, body: unknown) => request<T>('PATCH', path, body),
  del: (path: string) => request<void>('DELETE', path),
};
```

```ts
import { defineStore } from 'pinia';
import { ref } from 'vue';
import { api } from '@/api/client';
import type { Tenant, User } from '@/api/types';

export const useAuthStore = defineStore('auth', () => {
  const user = ref<User | null>(null);
  const tenant = ref<Tenant | null>(null);
  const ready = ref(false);

  async function fetchMe() {
    try {
      const me = await api.get<{ user: User; tenant: Tenant }>('/api/auth/me');
      user.value = me.user; tenant.value = me.tenant;
    } catch {
      user.value = null; tenant.value = null;
    } finally {
      ready.value = true;
    }
  }
  async function login(payload: { tenantSlug: string; email: string; password: string }) {
    const me = await api.post<{ user: User; tenant: Tenant }>('/api/auth/login', payload);
    user.value = me.user; tenant.value = me.tenant; ready.value = true;
  }
  async function register(payload: { tenant: { slug: string; name: string; countryCode: 'mk' }; user: { email: string; password: string; firstName: string; lastName: string } }) {
    const me = await api.post<{ user: User; tenant: Tenant }>('/api/auth/register', payload);
    user.value = me.user; tenant.value = me.tenant; ready.value = true;
  }
  async function logout() {
    await api.post('/api/auth/logout');
    user.value = null; tenant.value = null;
  }
  return { user, tenant, ready, fetchMe, login, register, logout };
```

`apps/web/src/composables/useResource.ts`:
```ts
import { ref } from 'vue';
import { api, ApiError } from '@/api/client';

type Query = Record<string, string | number | boolean | undefined>;

export function useResource<T extends { id: string }>(path: string) {
  const rows = ref<T[]>([]) as import('vue').Ref<T[]>;
  const total = ref(0);
  const loading = ref(false);
  const error = ref<string | null>(null);
  let lastQuery: Query | undefined;

  async function load(query?: Query) {
    loading.value = true; error.value = null; lastQuery = query;
    try {
      const res = await api.get<{ rows: T[]; total?: number }>(path, query);
      rows.value = res.rows; total.value = res.total ?? res.rows.length;
    } catch (e) { error.value = e instanceof ApiError ? e.code : 'error'; throw e; } finally { loading.value = false; }
  }
  async function wrap<R>(fn: () => Promise<R>): Promise<R> {
    error.value = null;
    try { const r = await fn(); await load(lastQuery); return r; } catch (e) { error.value = e instanceof ApiError ? e.code : 'error'; throw e; }
  }
  const create = (body: unknown) => wrap(() => api.post<T>(path, body));
  const update = (id: string, patch: unknown) => wrap(() => api.patch<T>(`${path}/${id}`, patch));
  const remove = (id: string) => wrap(() => api.del(`${path}/${id}`));
  return { rows, total, loading, error, load, create, update, remove };
}
```
